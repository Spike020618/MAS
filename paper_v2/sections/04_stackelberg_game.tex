%======================================================================
\section{Stackelberg Consensus Game}
\label{sec:game}
%======================================================================

\subsection{Game Formulation}

We formulate multi-agent consensus as a Stackelberg game $\mathcal{G} = (\mathcal{L}, \mathcal{F}, \Theta, \mathcal{S}, U_L, U_F)$.

\paragraph{Players.} The \emph{leader} $\mathcal{L}$ is the mechanism designer (bilevel optimizer). The \emph{followers} $\mathcal{F} = \{1, \ldots, n\}$ are the LLM agents.

\paragraph{Strategy Spaces.} The leader chooses mechanism parameters $\theta \in \Theta \subset \mathbb{R}^{16}$, controlling how AEIC quality signals map to consensus weights. Each follower $i$ chooses a strategy $s_i \in \{s^{\text{ind}}, s^{\text{syc}}\}$:
\begin{itemize}
    \item $s^{\text{ind}}$ (independent reasoning): produces high IC, EC, EQ at cost $C_{\text{ind}}$;
    \item $s^{\text{syc}}$ (sycophantic): produces lower IC/EC but potentially high EV at cost $C_{\text{syc}} < C_{\text{ind}}$.
\end{itemize}

\paragraph{Timing.}
\begin{enumerate}
    \item Leader commits to parameters $\theta$ (mechanism rules).
    \item Each follower observes $\theta$ and chooses strategy $s_i$.
    \item Agents produce AEIC outputs; quality signals $\tau_i$ are computed.
    \item Mechanism computes weights $w_i(\tau_i; \theta)$ via the scoring rule.
    \item Weighted voting produces the consensus decision.
\end{enumerate}

\paragraph{Utility Functions.} The follower utility is:
\begin{equation}
    U_i(s_i; \theta) = w_i(\tau_i(s_i); \theta) \cdot R - C(s_i)
    \label{eq:follower_utility}
\end{equation}
where $R > 0$ is the reward for adoption. The leader maximizes:
\begin{equation}
    \max_{\theta \in \Theta} \; \text{Accuracy}(\theta) + \lambda \cdot \text{IC}(\theta)
    \label{eq:leader_obj}
\end{equation}

\subsection{Scoring Rule}
\label{sec:scoring}

The 16 parameters $\theta$ control two aspects:

\begin{itemize}
    \item \textbf{Bid construction} (12 parameters): How IC, EC, EQ, EV signals combine into agent ``bids'' (alignment willingness $\alpha_i$, revision cost, historical alignment $h_i$).
    \item \textbf{Stackelberg dynamics} (4 parameters): Consensus energy coefficient $\lambda_c$, budget penalty $\mu$, learning rate $\eta$, and peer pressure $\gamma$.
\end{itemize}

Specifically, for agent $i$:
\begin{align}
    \alpha_i &= \max\!\big(\theta_{\text{floor}}, \; \text{IC}_i^{\theta_{\text{ic}}} \cdot \text{EQ}_i^{\theta_{\text{eq}}} \cdot (\theta_{\text{ev\_base}} + \theta_{\text{ev\_scale}} \cdot \text{EV}_i)\big) \\
    h_i &= \theta_{h_1} \cdot \text{IC}_i + \theta_{h_2} \cdot \text{EC}_i + \theta_{h_3} \cdot \text{EQ}_i + \theta_{h_4} \cdot \text{EV}_i
\end{align}

These bids are processed through a distributed consensus game that produces final weights $w_i(\theta)$ via iterative updates incorporating peer pressure from a graph Laplacian defined by the communication topology.

\subsection{Incentive Compatibility Metric}
\label{sec:ic_metric}

We define incentive compatibility through inter-group weight comparison:

\begin{definition}[Incentive Compatibility Metric]
For each question $q$, rank agents by quality score $\phi(\tau_i) = (\text{IC}_i + \text{EC}_i + \text{EQ}_i)/3$. Let $H_q$ = top-2 agents and $L_q$ = bottom-2 agents. Define:
\begin{equation}
    \text{IC}(\theta) = \frac{1}{|Q|} \sum_{q \in Q} \mathbb{1}\!\left[\frac{1}{|H_q|}\sum_{i \in H_q} w_i(\theta) > \frac{1}{|L_q|}\sum_{j \in L_q} w_j(\theta)\right]
    \label{eq:ic_metric}
\end{equation}
\end{definition}

This measures the fraction of questions where higher-quality reasoning receives greater consensus weight, capturing the essence of incentive compatibility.

\subsection{Bilevel Optimization}

The Stackelberg game is solved via bilevel optimization:
\begin{align}
    \textbf{Upper (Leader):} \quad & \max_{\theta \in \Theta} \; \text{Accuracy}(\theta) + \lambda \cdot \text{IC}(\theta) \label{eq:upper} \\
    \textbf{Lower (Followers):} \quad & w_i = f(\tau_i; \theta) \quad \forall i \label{eq:lower}
\end{align}

The lower-level problem is deterministic given $\theta$ (agents' AEIC outputs are fixed offline), reducing to single-level optimization implementable via differential evolution~\cite{storn1997differential}. Per-topology $\theta^*_T$ is optimized with population size 16 (1$\times$ the number of parameters) and 30 generations, then evaluated via 5-fold stratified cross-validation to prevent train/test leakage.

\subsection{Online Stackelberg Extension}
\label{sec:online_sg}

In multi-round topologies (debate, hierarchical), the mechanism operates \emph{online}: between rounds $t$ and $t{+}1$, it computes interim AEIC signals from round-$t$ outputs, runs the Stackelberg game to obtain current weights $w_i^{(t)}$, and injects quality feedback into agent prompts:
\begin{equation}
    \text{context}_i^{(t+1)} = \text{context}_i^{(t+1)} \cup \{(\text{IC}_i^{(t)}, \text{EC}_i^{(t)}, w_i^{(t)})\}
\end{equation}
This transforms the game from post-hoc reweighting to real-time incentive shaping: agents observe their current quality assessment and can adjust reasoning effort accordingly. The star topology is inherently online---leaf trust scores computed via Stackelberg are injected into the hub's synthesis context.
